{{- /* Numeric compare of exact Flix pins a vs b: -1, 0, or 1. */ -}} {{- $a := split .a "." -}} {{- $b := split .b "." -}} {{- $n := len $a -}} {{- if gt (len $b) $n }}{{ $n = len $b }}{{ end -}} {{- $result := 0 -}} {{- range seq 0 (sub $n 1) -}} {{- if eq $result 0 -}} {{- $left := 0 -}} {{- if lt . (len $a) }}{{ $left = index $a . | int }}{{ end -}} {{- $right := 0 -}} {{- if lt . (len $b) }}{{ $right = index $b . | int }}{{ end -}} {{- if lt $left $right }}{{ $result = -1 }}{{ else if gt $left $right }}{{ $result = 1 }}{{ end -}} {{- end -}} {{- end -}} {{- return $result -}}
